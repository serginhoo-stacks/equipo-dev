```javascript // ============================================================ // SMARTINSPECT AI — AUTENTICAÇÃO // ============================================================ const supabaseClient = window.supabaseClient || window.supabase; // ------------------------------------------------------------ // ELEMENTOS DA PÁGINA // ------------------------------------------------------------ const loginForm = document.getElementById("loginForm"); if (loginForm) { loginForm.hidden = false; } if (resetRequestForm) { resetRequestForm.hidden = true; } if (resetVerifyForm) { resetVerifyForm.hidden = true; } if (resetPasswordForm) { resetPasswordForm.hidden = true; } if (authAccountLinks) { authAccountLinks.hidden = false; } // Volta para a primeira etapa if (emailStep) { emailStep.hidden = false; } if (passwordStep) { passwordStep.hidden = true; } if (senhaInput) { senhaInput.value = ""; senhaInput.required = false; } } // ------------------------------------------------------------ // MOSTRAR RECUPERAÇÃO — SOLICITAR CÓDIGO // ------------------------------------------------------------ function showResetRequest() { hideAuthMessage(); if (loginForm) { loginForm.hidden = true; } if (resetRequestForm) { resetRequestForm.hidden = false; } if (resetVerifyForm) { resetVerifyForm.hidden = true; } if (resetPasswordForm) { resetPasswordForm.hidden = true; } if (authAccountLinks) { authAccountLinks.hidden = true; } if (resetEmailInput && emailInput) { resetEmailInput.value = emailInput.value.trim(); } if (resetEmailInput) { resetEmailInput.focus(); } } // ------------------------------------------------------------ // MOSTRAR VERIFICAÇÃO DO CÓDIGO // ------------------------------------------------------------ function showResetVerify(email) { hideAuthMessage(); if (loginForm) { loginForm.hidden = true; } if (resetRequestForm) { resetRequestForm.hidden = true; } if (resetVerifyForm) { resetVerifyForm.hidden = false; } if (resetPasswordForm) { resetPasswordForm.hidden = true; } if (authAccountLinks) { authAccountLinks.hidden = true; } if (verifyEmailInput) { verifyEmailInput.value = email; } if (resetCodeInput) { resetCodeInput.value = ""; resetCodeInput.focus(); } } // ------------------------------------------------------------ // MOSTRAR NOVA SENHA // ------------------------------------------------------------ function showNewPassword() { hideAuthMessage(); if (loginForm) { loginForm.hidden = true; } if (resetRequestForm) { resetRequestForm.hidden = true; } if (resetVerifyForm) { resetVerifyForm.hidden = true; } if (resetPasswordForm) { resetPasswordForm.hidden = false; } if (authAccountLinks) { authAccountLinks.hidden = true; } if (newPasswordInput) { newPasswordInput.focus(); } } // ============================================================ // LOGIN — ETAPA 1 // VERIFICAR E-MAIL // ============================================================ if (verifyEmailButton) { verifyEmailButton.addEventListener("click", async () => { const email = emailInput.value.trim().toLowerCase(); hideAuthMessage(); // ------------------------- // Validação // ------------------------- if (!email) { showAuthMessage( "Digite seu e-mail.", "error" ); emailInput.focus(); return; } if (!emailInput.checkValidity()) { showAuthMessage( "Digite um e-mail válido.", "error" ); emailInput.focus(); return; } // ------------------------- // Loading // ------------------------- verifyEmailButton.disabled = true; const originalText = verifyEmailButton.innerHTML; verifyEmailButton.textContent = "Verificando..."; try { /* * IMPORTANTE * * Esta URL deve apontar para uma Edge Function * criada no seu projeto Supabase. * * Exemplo: * * /functions/v1/check-email */ const response = await fetch( "/functions/v1/check-email", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: email }) } ); const result = await response.json(); // ------------------------- // E-mail não encontrado // ------------------------- if ( !response.ok || !result.exists ) { showAuthMessage( "Não encontramos uma conta com esse e-mail.", "error" ); return; } // ------------------------- // E-mail encontrado // ------------------------- showAuthMessage( "E-mail verificado. Agora digite sua senha.", "success" ); emailInput.value = email; // Esconde e-mail emailStep.hidden = true; // Mostra senha passwordStep.hidden = false; senhaInput.required = true; senhaInput.focus(); } catch (error) { console.error( "Erro ao verificar e-mail:", error ); showAuthMessage( "Não foi possível verificar o e-mail. Tente novamente.", "error" ); } finally { verifyEmailButton.disabled = false; verifyEmailButton.innerHTML = originalText; } }); } // ============================================================ // LOGIN — ETAPA 2 // E-MAIL + SENHA // ============================================================ if (loginForm) { loginForm.addEventListener("submit", async (event) => { event.preventDefault(); hideAuthMessage(); const email = emailInput.value.trim().toLowerCase(); const password = senhaInput.value; // ------------------------- // Validação // ------------------------- if (!email) { showAuthMessage( "Digite seu e-mail.", "error" ); return; } if (!password) { showAuthMessage( "Digite sua senha.", "error" ); senhaInput.focus(); return; } // ------------------------- // Loading // ------------------------- const submitButton = loginForm.querySelector( 'button[type="submit"]' ); const originalText = submitButton ? submitButton.innerHTML : ""; if (submitButton) { submitButton.disabled = true; submitButton.textContent = "Entrando..."; } try { // ------------------------- // Login Supabase // ------------------------- const { data, error } = await supabaseClient.auth.signInWithPassword({ email: email, password: password }); // ------------------------- // Erro // ------------------------- if (error) { console.error( "Erro no login:", error ); showAuthMessage( "E-mail ou senha incorretos.", "error" ); return; } // ------------------------- // Login realizado // ------------------------- if (data && data.session) { showAuthMessage( "Login realizado com sucesso!", "success" ); // Pequeno atraso para mostrar mensagem setTimeout(() => { window.location.href = "./dashboard.html"; }, 500); } } catch (error) { console.error(error); showAuthMessage( "Ocorreu um erro ao entrar. Tente novamente.", "error" ); } finally { if (submitButton) { submitButton.disabled = false; submitButton.innerHTML = originalText; } } }); } // ============================================================ // ESQUECI MINHA SENHA // ============================================================ if (resetPasswordLink) { resetPasswordLink.addEventListener( "click", (event) => { event.preventDefault(); showResetRequest(); } ); } // ============================================================ // SOLICITAR CÓDIGO DE RECUPERAÇÃO // ============================================================ if (resetRequestForm) { resetRequestForm.addEventListener( "submit", async (event) => { event.preventDefault(); hideAuthMessage(); const email = resetEmailInput.value .trim() .toLowerCase(); if (!email) { showAuthMessage( "Digite o e-mail da sua conta.", "error" ); resetEmailInput.focus(); return; } if (!resetEmailInput.checkValidity()) { showAuthMessage( "Digite um e-mail válido.", "error" ); return; } const button = resetRequestForm.querySelector( 'button[type="submit"]' ); if (button) { button.disabled = true; button.textContent = "Enviando..."; } try { /* * Sua Edge Function deve: * * 1. receber o e-mail * 2. gerar o código * 3. salvar o código com validade * 4. enviar o código por e-mail */ const response = await fetch( "/functions/v1/request-password-reset", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: email }) } ); const result = await response.json(); if (!response.ok) { throw new Error( result.message || "Erro ao solicitar código." ); } showAuthMessage( "Código enviado! Verifique seu e-mail.", "success" ); verifyEmailInput.value = email; setTimeout(() => { showResetVerify(email); }, 700); } catch (error) { console.error(error); showAuthMessage( error.message || "Não foi possível enviar o código.", "error" ); } finally { if (button) { button.disabled = false; button.textContent = "Enviar código"; } } } ); } // ============================================================ // VERIFICAR CÓDIGO // ============================================================ if (resetVerifyForm) { resetVerifyForm.addEventListener( "submit", async (event) => { event.preventDefault(); hideAuthMessage(); const email = verifyEmailInput.value .trim() .toLowerCase(); const code = resetCodeInput.value .trim(); // ------------------------- // Validação // ------------------------- if (!/^\d{8}$/.test(code)) { showAuthMessage( "Digite o código de 8 dígitos.", "error" ); resetCodeInput.focus(); return; } const button = resetVerifyForm.querySelector( 'button[type="submit"]' ); if (button) { button.disabled = true; button.textContent = "Verificando..."; } try { const response = await fetch( "/functions/v1/verify-password-reset", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: email, code: code }) } ); const result = await response.json(); if ( !response.ok || !result.valid ) { throw new Error( result.message || "Código inválido ou expirado." ); } showAuthMessage( "Código confirmado! Crie sua nova senha.", "success" ); // Guarda um token temporário // retornado pela Edge Function if (result.resetToken) { sessionStorage.setItem( "smartinspect_reset_token", result.resetToken ); } setTimeout(() => { showNewPassword(); }, 700); } catch (error) { console.error(error); showAuthMessage( error.message || "Código inválido ou expirado.", "error" ); } finally { if (button) { button.disabled = false; button.textContent = "Verificar código"; } } } ); } // ============================================================ // TROCAR SENHA // ============================================================ if (resetPasswordForm) { resetPasswordForm.addEventListener( "submit", async (event) => { event.preventDefault(); hideAuthMessage(); const password = newPasswordInput.value; const confirmPassword = confirmNewPasswordInput.value; // ------------------------- // Validações // ------------------------- if (password.length < 8) { showAuthMessage( "A nova senha precisa ter pelo menos 8 caracteres.", "error" ); newPasswordInput.focus(); return; } if (password !== confirmPassword) { showAuthMessage( "As senhas não são iguais.", "error" ); confirmNewPasswordInput.focus(); return; } const resetToken = sessionStorage.getItem( "smartinspect_reset_token" ); if (!resetToken) { showAuthMessage( "Sua sessão de recuperação expirou. Solicite um novo código.", "error" ); return; } const button = resetPasswordForm.querySelector( 'button[type="submit"]' ); if (button) { button.disabled = true; button.textContent = "Alterando senha..."; } try { const response = await fetch( "/functions/v1/reset-password", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ token: resetToken, password: password }) } ); const result = await response.json(); if (!response.ok) { throw new Error( result.message || "Não foi possível alterar a senha." ); } // Apaga token sessionStorage.removeItem( "smartinspect_reset_token" ); showAuthMessage( "Senha alterada com sucesso! Agora você pode entrar.", "success" ); // Limpa campos newPasswordInput.value = ""; confirmNewPasswordInput.value = ""; setTimeout(() => { showLogin(); }, 1500); } catch (error) { console.error(error); showAuthMessage( error.message || "Não foi possível alterar a senha.", "error" ); } finally { if (button) { button.disabled = false; button.textContent = "Trocar senha"; } } } ); } // ============================================================ // REENVIAR CÓDIGO // ============================================================ if (resendResetCode) { resendResetCode.addEventListener( "click", async () => { const email = verifyEmailInput.value .trim() .toLowerCase(); if (!email) { showAuthMessage( "Informe seu e-mail novamente.", "error" ); return; } resendResetCode.disabled = true; const originalText = resendResetCode.textContent; resendResetCode.textContent = "Enviando..."; try { const response = await fetch( "/functions/v1/request-password-reset", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email: email }) } ); const result = await response.json(); if (!response.ok) { throw new Error( result.message || "Erro ao reenviar código." ); } showAuthMessage( "Um novo código foi enviado para seu e-mail.", "success" ); resetCodeInput.value = ""; } catch (error) { console.error(error); showAuthMessage( error.message || "Não foi possível reenviar o código.", "error" ); } finally { resendResetCode.disabled = false; resendResetCode.textContent = originalText; } } ); } // ============================================================ // VOLTAR PARA LOGIN // ============================================================ if (backToLoginFromRequest) { backToLoginFromRequest.addEventListener( "click", () => { showLogin(); } ); } // ============================================================ // TROCAR E-MAIL // ============================================================ if (backToResetRequest) { backToResetRequest.addEventListener( "click", () => { showResetRequest(); } ); } // ============================================================ // CÓDIGO — ACEITAR SOMENTE NÚMEROS // ============================================================ if (resetCodeInput) { resetCodeInput.addEventListener( "input", () => { resetCodeInput.value = resetCodeInput.value .replace(/\D/g, "") .slice(0, 8); } ); } // ============================================================ // VERIFICAR SE JÁ ESTÁ LOGADO // ============================================================ async function checkExistingSession() { try { const { data, error } = await supabaseClient.auth.getSession(); if (error) { console.error(error); return; } if ( data && data.session ) { window.location.href = "./dashboard.html"; } } catch (error) { console.error( "Erro ao verificar sessão:", error ); } } // ============================================================ // INICIALIZAÇÃO // ============================================================ document.addEventListener( "DOMContentLoaded", () => { showLogin(); checkExistingSession(); } ); ```
